cmake_minimum_required(VERSION 3.20)
project(NeonRun LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "NEON RUN currently uses the native Windows graphics and audio APIs.")
endif()

add_executable(NeonRun WIN32 main.cpp)
target_compile_features(NeonRun PRIVATE cxx_std_17)
target_compile_definitions(NeonRun PRIVATE UNICODE _UNICODE)
target_link_libraries(NeonRun PRIVATE winmm)
if(MSVC)
  target_compile_options(NeonRun PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(NeonRun PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(NeonRun PRIVATE -municode)
endif()
